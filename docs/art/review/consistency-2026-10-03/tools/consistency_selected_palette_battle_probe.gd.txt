extends SceneTree
const OUT := "C:/Users/pmcga/Documents/Codex/2026-10-01/referenced-chatgpt-conversation-this-is-an/consistency-review/"
var battle: Node
var fixture: Node
var observing := false
var tag := ""
var seen := {}
var captures: Array = []
var checks: Array = []
func _initialize() -> void:
	call_deferred("_run")
func _capture(name: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT+name+".png")
func _watch() -> void:
	while observing:
		await process_frame
		if not is_instance_valid(battle):
			continue
		for side in ["_foe_sprite","_player_sprite"]:
			var sprite: Sprite2D = battle.get(side)
			var mat: ShaderMaterial = sprite.material
			if mat == null or not (mat.get_shader_parameter("keep_authored_cracks") or mat.get_shader_parameter("charcoal_head")):
				continue
			var path := sprite.texture.resource_path
			var key: String = tag+side+path
			if seen.has(key):
				continue
			seen[key] = true
			var name: String = tag+"-"+side.trim_prefix("_")+"-"+path.get_file().get_basename()
			await _capture(name)
			captures.append({"name":name,"texture":path,"side":side,"frame":Engine.get_process_frames(),"cracks":mat.get_shader_parameter("keep_authored_cracks"),"charcoal_head":mat.get_shader_parameter("charcoal_head"),"heat":mat.get_shader_parameter("heat")})
		for band in battle.get_children():
			if not band is ColorRect: continue
			for child in band.get_children():
				if not child is Sprite2D or not child.material is ShaderMaterial: continue
				var portrait: Sprite2D = child
				if not (portrait.material.get_shader_parameter("keep_authored_cracks") or portrait.material.get_shader_parameter("charcoal_head")) or seen.has(tag+"portrait"): continue
				# Wait for the close-up to finish its entrance before capturing it.
				if absf(portrait.position.x - 576.0) > 350.0: continue
				seen[tag+"portrait"] = true
				await _capture(tag+"-signature-portrait")
				captures.append({"name":tag+"-signature-portrait","texture":portrait.texture.resource_path,"cracks":portrait.material.get_shader_parameter("keep_authored_cracks"),"charcoal_head":portrait.material.get_shader_parameter("charcoal_head"),"frame":Engine.get_process_frames()})

func _run() -> void:
	root.size = Vector2i(1152,648)
	fixture = Node.new()
	fixture.process_mode = Node.PROCESS_MODE_ALWAYS
	root.add_child(fixture)
	current_scene = fixture
	var script: Script = load("res://scripts/battle_screen.gd")
	battle = CanvasLayer.new()
	battle.set_script(script)
	fixture.add_child(battle)
	var vitrecrag: Resource = load("res://data/creatures/108_vitrecrag.tres")
	var cinderhide: Resource = load("res://data/creatures/002_cinderhide.tres")
	var ironvent_attack: Resource = load("res://data/creatures/102_ironvent.tres")
	var clinkerhide: Resource = load("res://data/creatures/101_clinkerhide.tres")
	for pair in [
		["selected-rear-brow-shear", clinkerhide, cinderhide, true, 0],
		["selected-rear-slagburst", clinkerhide, cinderhide, true, 3]]:
		battle = await _new_battle()
		tag = pair[0]
		root.get_node("PlayerParty").caught.clear()
		root.get_node("PlayerParty").active_slot = -1
		battle.player_creature = pair[1]
		await battle.start(pair[2])
		await _capture(tag+"-before")
		var side := "_player_sprite" if pair[3] else "_foe_sprite"
		var caster: Sprite2D = battle.get(side)
		var original: Texture2D = caster.texture
		var home: Vector2 = caster.position
		observing = true
		_watch()
		await battle._attack(pair[1] if pair[3] else pair[2],pair[3],(pair[1] if pair[3] else pair[2]).moves[pair[4]])
		observing = false
		for i in 360: await process_frame
		await _capture(tag+"-after")
		var material: ShaderMaterial = caster.material
		var result := {"name":tag,"cracks_released":not material.get_shader_parameter("keep_authored_cracks"),"texture_restored":caster.texture == original,"position_restored":caster.position.distance_to(home)<0.1,"reversed_released":not battle.get("_reversed"),"head_finish_owner":material.get_shader_parameter("charcoal_head")==((pair[1] if pair[3] else pair[2]).id==102),"player_hp":battle.get("_player_hp"),"foe_hp":battle.get("_foe_hp")}
		checks.append(result)
		print("ATTACK_CHECK ",result)
		paused = false
	var roster: Array = []
	for file in DirAccess.get_files_at("res://data/creatures"):
		if not file.ends_with(".tres"): continue
		battle = await _new_battle()
		var creature: Resource = load("res://data/creatures/"+file)
		root.get_node("PlayerParty").caught.clear()
		root.get_node("PlayerParty").active_slot = -1
		battle.player_creature = creature
		await battle.start(creature)
		await _capture("roster-"+file.trim_suffix(".tres"))
		roster.append({"name":creature.creature_name,"id":creature.id,"front":creature.sprite.resource_path,"back":creature.back_sprite.resource_path if creature.back_sprite else "","battle_size":creature.battle_size,"front_scale":str(battle.get("_foe_sprite").scale),"back_scale":str(battle.get("_player_sprite").scale)})
		paused = false
	var ironvent: Resource = load("res://data/creatures/102_ironvent.tres")
	var clean: Image = ironvent.back_sprite.get_image()
	clean.save_png(OUT+"ironvent-rear-native-region.png")
	checks.append({"name":"Ironvent rear region","size":str(ironvent.back_sprite.get_size()),"image_size":str(clean.get_size()),"used_rect":str(clean.get_used_rect()),"region_excludes_row96":clean.get_height()==96 and ironvent.back_sprite.get_height()==100})
	var file := FileAccess.open(OUT+"selected-palette-battle-results.json",FileAccess.WRITE)
	file.store_string(JSON.stringify({"checks":checks,"captures":captures,"roster":roster},"\t"))
	print("CONSISTENCY_RUNTIME_COMPLETE ",roster.size())
	battle.queue_free()
	for i in 5: await process_frame
	var failed := false
	for check in checks:
		for key in ["cracks_released","texture_restored","position_restored","reversed_released","head_finish_owner","region_excludes_row96"]:
			if check.has(key) and not check[key]: failed = true
	print("FINAL_BATTLE_CHECKS_PASSED ",not failed)
	quit(1 if failed else 0)

func _new_battle() -> Node:
	if is_instance_valid(battle):
		battle.queue_free()
		paused = false
		for i in 5: await process_frame
	var instance := CanvasLayer.new()
	instance.set_script(load("res://scripts/battle_screen.gd"))
	fixture.add_child(instance)
	return instance
