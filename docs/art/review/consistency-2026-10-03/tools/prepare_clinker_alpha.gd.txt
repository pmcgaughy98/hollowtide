extends SceneTree
func _initialize() -> void:
	var source := "C:/Users/pmcga/Documents/Codex/2026-10-01/referenced-chatgpt-conversation-this-is-an/consistency-review/clinkerhide-user-selected-transparent.png"
	var output := "C:/Users/pmcga/Documents/Codex/2026-10-01/referenced-chatgpt-conversation-this-is-an/consistency-review/clinkerhide-selected-back-native100.png"
	var image := Image.load_from_file(source)
	if image == null or image.get_size() != Vector2i(100, 100):
		push_error("Chosen rear has an unexpected canvas")
		quit(1)
		return
	image.convert(Image.FORMAT_RGBA8)
	var removed: Array = []
	for y in image.get_height():
		for x in image.get_width():
			var color := image.get_pixel(x, y)
			if color.a > 0.0 and color.r > 0.70 and color.g < 0.28 and color.b > 0.70:
				removed.append(Vector2i(x, y))
				color.a = 0.0
				image.set_pixel(x, y, color)
	if removed.size() != 2:
		push_error("Unexpected remaining key-color count")
		quit(1)
		return
	var error := image.save_png(output)
	print("CLINKERHIDE_ALPHA_PREP ", JSON.stringify({"removed_background_pixels":str(removed),"size":str(image.get_size()),"used_rect":str(image.get_used_rect()),"save_error":error}))
	quit(0 if error == OK else 1)
