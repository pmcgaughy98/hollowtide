extends SceneTree
const OUT := "C:/Users/pmcga/Documents/Codex/2026-10-01/referenced-chatgpt-conversation-this-is-an/consistency-review/"
var menu: Node
var checks: Array = []
var failed := false
func _initialize() -> void:
	call_deferred("_run")
func _frames(n: int) -> void:
	for i in n: await process_frame
func _key(code: Key) -> void:
	var e := InputEventKey.new()
	e.keycode = code
	e.pressed = true
	Input.parse_input_event(e)
	await _frames(2)
	e = InputEventKey.new()
	e.keycode = code
	Input.parse_input_event(e)
	await _frames(2)
func _pad(button: JoyButton) -> void:
	for pressed in [true,false]:
		var event := InputEventJoypadButton.new()
		event.button_index = button
		event.pressed = pressed
		event.device = 0
		Input.parse_input_event(event)
		await _frames(4)
func _click(control: Control) -> void:
	var at := control.get_global_rect().get_center()
	var motion := InputEventMouseMotion.new()
	motion.position = at
	motion.global_position = at
	Input.parse_input_event(motion)
	for pressed in [true,false]:
		var e := InputEventMouseButton.new()
		e.position = at
		e.global_position = at
		e.button_index = MOUSE_BUTTON_LEFT
		e.pressed = pressed
		Input.parse_input_event(e)
		await _frames(3)
func _check(name: String, pass_: bool, data: Variant=null) -> void:
	checks.append({"name":name,"passed":pass_,"data":data})
	failed = failed or not pass_
	print("SELECT_CHECK ",name," ",pass_," ",data)
func _selected(model: String) -> void:
	var other := "Male" if model == "Female" else "Female"
	var cards: Array = menu.get("_courier_buttons")
	for card in cards:
		var chosen: bool = card.name == model
		_check(str(card.name)+" caption for "+model,card.get_node("Caption").text == str(card.name)+(" (selected)" if chosen else ""),card.get_node("Caption").text)
		_check(str(card.name)+" toggle for "+model,card.button_pressed == chosen)
	_check("model value "+model,menu.get("_selected_courier")==model)
func _capture(name: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT+"selection-"+name+".png")
func _run() -> void:
	var scene: PackedScene = load("res://scenes/main_menu_art_preview.tscn")
	menu = scene.instantiate()
	root.add_child(menu)
	current_scene = menu
	await _frames(150)
	await _key(KEY_ENTER)
	await _frames(5)
	var cards: Array = menu.get("_courier_buttons")
	await _click(cards[0])
	_selected("Male")
	await _key(KEY_RIGHT)
	_selected("Female")
	await _capture("female-arrow")
	await _key(KEY_LEFT)
	_selected("Male")
	await _capture("male-arrow")
	await _pad(JOY_BUTTON_DPAD_RIGHT)
	_selected("Female")
	await _capture("female-pad")
	await _pad(JOY_BUTTON_DPAD_LEFT)
	_selected("Male")
	await _click(cards[1])
	_selected("Female")
	await _capture("female-mouse")
	await _key(KEY_DOWN)
	await _key(KEY_ENTER)
	await _frames(150)
	var player := current_scene.get_node_or_null("Player")
	_check("female entered route",player!=null and player.get("chibi_model")=="Female")
	if player!=null:
		var sprite: Sprite2D=player.get_node("Sprite2D")
		_check("female finish installed",sprite.material is ShaderMaterial and sprite.material.shader.resource_path=="res://scripts/effects/courier_finish.gdshader")
		var clerk: Sprite2D = current_scene.get_node("RouteDecor/CityCourierClerk")
		_check("city clerk uses same female finish",clerk.material is ShaderMaterial and clerk.material.shader.resource_path == sprite.material.shader.resource_path)
		var start: Vector2=player.position
		Input.action_press("move_right")
		for i in 12: await physics_frame
		Input.action_release("move_right")
		await _frames(3)
		_check("ordinary walking",player.position.x>start.x+5,{"from":str(start),"to":str(player.position)})
		await _capture("female-route")
	var f:=FileAccess.open(OUT+"selection-final-with-clerk-results.json",FileAccess.WRITE)
	f.store_string(JSON.stringify({"checks":checks,"failed":failed},"\t"))
	print("SELECTION_COMPLETE ",not failed)
	quit(1 if failed else 0)
