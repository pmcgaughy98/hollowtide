extends SceneTree
const OUT := "C:/Users/pmcga/Documents/Codex/2026-10-01/referenced-chatgpt-conversation-this-is-an/consistency-review/"
func _initialize() -> void:
	call_deferred("_run")
func _run() -> void:
	root.size = Vector2i(1152, 648)
	var bg := ColorRect.new()
	bg.color = Color("#211d35")
	bg.size = Vector2(1152, 648)
	root.add_child(bg)
	var shader: Shader = load("C:/Users/pmcga/Documents/Codex/2026-10-01/referenced-chatgpt-conversation-this-is-an/courier_finish.gdshader")
	var base := "res://assets/sprites/player/retro_diffusion_chibi_walking_2026_10_01/"
	var male: Texture2D = load(base + "male_chibi_walk_matched_turns_native80.png")
	var female: Texture2D = load(base + "female_chibi_walk_eight_directions_native80.png")
	for row in 3:
		for col in 8:
			var label := Label.new()
			label.text = ["Male", "Female before", "Female finish"][row] + " " + str(col)
			label.position = Vector2(col * 144 + 5, row * 210 + 4)
			label.add_theme_font_size_override("font_size", 13)
			root.add_child(label)
			var sprite := Sprite2D.new()
			sprite.texture = male if row == 0 else female
			sprite.hframes = 8
			sprite.vframes = 8
			sprite.frame = col * 8
			sprite.position = Vector2(col * 144 + 72, row * 210 + 117)
			sprite.scale = Vector2.ONE * 2
			sprite.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
			if row == 2:
				var mat := ShaderMaterial.new()
				mat.shader = shader
				sprite.material = mat
			root.add_child(sprite)
	for i in 10:
		await process_frame
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT + "courier-finish-comparison.png")
	var view := SubViewport.new()
	view.size = Vector2i(640,640)
	view.transparent_bg = true
	view.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(view)
	var full := Sprite2D.new()
	full.texture = female
	full.position = Vector2(320,320)
	full.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	var mat := ShaderMaterial.new()
	mat.shader = shader
	full.material = mat
	view.add_child(full)
	for i in 5:
		await process_frame
	await RenderingServer.frame_post_draw
	view.get_texture().get_image().save_png(OUT + "female-finish-all64-render.png")
	print("FINISH_RENDER_OK")
	quit()
