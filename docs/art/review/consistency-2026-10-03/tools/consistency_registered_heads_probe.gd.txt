extends SceneTree
const OUT := "C:/Users/pmcga/Documents/Codex/2026-10-01/referenced-chatgpt-conversation-this-is-an/consistency-review/"
func _initialize() -> void:
	call_deferred("_run")
func _run() -> void:
	root.size = Vector2i(1080,1080)
	var background := ColorRect.new()
	background.color = Color("#211d35")
	background.size = Vector2(1080,1080)
	root.add_child(background)
	var owner := CanvasLayer.new()
	owner.set_script(load("res://scripts/battle_screen.gd"))
	var shader: Shader = load("res://scripts/effects/creature_sprite.gdshader")
	var views: Array = []
	var check: Array = []
	for name in ["rest","warm","pressure","rear","open","roar","settle","cool","titlecard_official_2026_09_30"]:
		var texture: Texture2D = load("res://assets/sprites/creatures/attacks/ironvent_furnace_roar/style_matched/"+name+".png")
		var native := SubViewport.new()
		native.size = Vector2i(100,100)
		native.transparent_bg = true
		native.render_target_update_mode = SubViewport.UPDATE_ALWAYS
		root.add_child(native)
		var sprite := Sprite2D.new()
		sprite.texture = texture
		sprite.position = Vector2(50,50)
		sprite.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
		var finish := ShaderMaterial.new()
		finish.shader = shader
		finish.set_shader_parameter("charcoal_head",true)
		var bounds: Vector4 = owner._ironvent_head_bounds(texture)
		finish.set_shader_parameter("head_finish_bounds",bounds)
		sprite.material = finish
		native.add_child(sprite)
		views.append([name,native])
		var display := Sprite2D.new()
		display.texture = texture
		display.material = finish
		display.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
		var n: int = views.size()-1
		display.position=Vector2((n%3)*360+158,(n/3)*360+185)
		display.scale = Vector2.ONE*3.0
		root.add_child(display)
		var label := Label.new()
		label.text = name
		label.position = Vector2((n%3)*360+8,(n/3)*360+8)
		root.add_child(label)
		check.append({"name":name,"bounds":str(bounds),"texture":texture.resource_path})
	for i in 6: await process_frame
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT+"ironvent-registered-head-poses.png")
	for pair in views:
		pair[1].get_texture().get_image().save_png(OUT+"ironvent-"+pair[0]+"-registered-head-native.png")
	var file := FileAccess.open(OUT+"ironvent-head-pose-registration.json",FileAccess.WRITE)
	file.store_string(JSON.stringify(check,"\t"))
	owner.free()
	print("REGISTERED_IRONVENT_HEAD_RENDER_COMPLETE")
	quit()
