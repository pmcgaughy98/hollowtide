extends SceneTree
const OUT := "C:/Users/pmcga/Documents/Codex/2026-10-01/referenced-chatgpt-conversation-this-is-an/consistency-review/"
func _initialize() -> void:
	call_deferred("_run")
func _run() -> void:
	var scene: PackedScene = load("res://scenes/chibi_walk_review.tscn")
	var review := scene.instantiate()
	root.add_child(review)
	current_scene = review
	var observed := {}
	for tick in 130:
		await process_frame
		for child in review.get_children():
			if child is AnimatedSprite2D:
				if not observed.has(child.name): observed[child.name] = []
				if child.frame not in observed[child.name]: observed[child.name].append(child.frame)
		if tick % 6 == 0 and tick >= 6:
			await RenderingServer.frame_post_draw
			root.get_texture().get_image().save_png(OUT+"walk-native-%03d.png" % tick)
	var checks: Array=[]
	for child in review.get_children():
		if child is AnimatedSprite2D:
			checks.append({"name":str(child.name),"eight_frames_observed":observed[child.name].size()==8,"frames":observed[child.name],"female_finish":child.material!=null if str(child.name).begins_with("Female") else child.material==null,"scale":str(child.scale)})
	var f:=FileAccess.open(OUT+"walk-runtime-results.json",FileAccess.WRITE)
	f.store_string(JSON.stringify(checks,"\t"))
	print("WALK_FINISH_REVIEW_COMPLETE ",checks.size())
	quit()
